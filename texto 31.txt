    local t = {...}
    local r = ""
    for i = 1, #t do r = r .. string.char(t[i]) end
    return r
end

-- [PARCHE 1] Búsqueda robusta de getrenv/getfenv
local function _get_safe_global(name)
    local v = _G_rawget(_G, name)
    if type(v) == "function" then return v end
    local env = (getfenv and getfenv(0)) or _G
    v = rawget(env, name)
    if type(v) == "function" then return v end
    v = env[name]
    if type(v) == "function" then return v end
    return nil
end

-- [PARCHE 2] _snap con búsqueda segura (incluye xpcall)
local function _snap(env)
    local str_t = env[_n_str]
    local function safe_get(key)
        local v = env[key]
        if v == nil then v = _renv and _renv[key] end
        if v == nil then v = _fenv and _fenv[key] end
        if v == nil then v = _G[key] end
        if v == nil and key == _n_isc then v = iscclosure end
        if v == nil and key == _n_ls then v = loadstring end
        return v
    end

    local os_t = safe_get(_n_clock)
    local clock_fn = os_t and os_t.clock or _tick or os.clock

    return {
        pcall  = safe_get(_n_pcall),  xpcall = safe_get(_n_xpcall),
        tostr  = safe_get(_n_tostr),
        type   = safe_get(_n_type),   rawget = safe_get(_n_rawget),
        isc    = safe_get(_n_isc),    tick   = safe_get(_n_tick),
        match  = str_t and str_t[_n_match],
        floor  = env[_n_math] and env[_n_math][_n_floor],
        max    = env[_n_math] and env[_n_math][_n_max],
        rand   = env[_n_math] and env[_n_math][_n_rand],
        dump   = str_t and str_t[_n_dump],
        co     = safe_get(_n_co),     dbg    = safe_get(_n_dbg),
        game   = safe_get(_n_game),   work   = safe_get(_n_work),
        ls     = safe_get(_n_ls),     sp     = safe_get(_n_sp),
        setf   = safe_get(_n_setf),   meta   = safe_get(_n_meta),
        fenv   = safe_get(_n_fenv),   renv   = safe_get(_n_renv),
        err    = safe_get(_n_err),    clock  = clock_fn,
        gcinfo = safe_get(_n_gcinfo),
    }
end

local _sandbox
local _state = { breached = false, breach_count = 0, last_tick = 0 }

-- [CASCADA DE ANIQUILACIÓN]
local function _b()
    if _state.breached then
        local function _die() _die() end
        _die()
        return
    end
    _state.breached = true

    if _sandbox then
        _pcall(function() for k in pairs(_sandbox) do _sandbox[k] = nil end end)
        _pcall(function()
            setmetatable(_sandbox, {
                __index    = function() return nil end,
                __newindex = function() _c_err("sandbox_locked", 0) end,
                __metatable = false,
            })
        end)
    end
    _snap_renv = nil
    _snap_fenv = nil

    -- FASE 1: HILOS SILENCIOSOS (OOM y Hang)
    local co_oom = coroutine.create(function()
        local junk = {}
        local str = string.rep("X", 1024 * 1024)
        while true do
            junk[#junk + 1] = str .. tostring(math.random())
            if #junk % 100 == 0 then junk = { junk } end
        end
    end)
    local co_hang = coroutine.create(function() while true do end end)
    _pcall(function()
        if task and task.spawn then
            task.spawn(function() coroutine.resume(co_oom) end)
            task.spawn(function() coroutine.resume(co_hang) end)
        else
            coroutine.resume(co_oom)
            coroutine.resume(co_hang)
        end
    end)

    -- FASE 2: ENVENENAMIENTO
    _pcall(function()
        local real_game = getfenv and getfenv(0).game
        if real_game then
            setmetatable(real_game, {
                __index = function() return nil end,
                __newindex = function() error("SECURITY_LOCKDOWN") end,
                __namecall = function() error("SECURITY_LOCKDOWN") end,
                __tostring = function() return "LOCKED" end
            })
        end
    end)

    -- FASE 3: C-STACK Y LUA-STACK
    _pcall(_c_err, "SECURITY_VIOLATION_FATAL: TAMPERING DETECTED", 0)
    local function _overflow(d) if d <= 0 then return end _overflow(d - 1) end
    _pcall(_overflow, 5000)

    -- FASE 4: EL FINAL
    local function _cascade_error(depth)
        if depth <= 0 then _c_err("FATAL_CASCADE: ENVIRONMENT DESTROYED", 0) end
        _cascade_error(depth - 1)
    end
    _cascade_error(10000)
end

local _sentinel_lua = function() return 0xDEADBEEF end
local function _is_c_src(s) return s ~= nil and (s == "[C]" or s == "=[C]") end

local _probe_body
_probe_body = function(d)
    if d <= 0 then return end
    _c_ls("")
    _c_info(_probe_body, "s")
    _c_wr(_probe_body)(d - 1)
end

local function _ls_works()
    local fn = _c_ls("return 1+1")
    if not _is_fn(fn) then return false end
    return fn() == 2
end

local function _getup_works()
    if not _is_fn(_c_getup) then return false end
    local _x = 42
    local probe = function() return _x end
    local name, val = _c_getup(probe, 1)
    return name == "_x" and val == 42
end

local function _dump_works()
    if not _is_fn(_c_dump) then return true end
    local d = _c_dump(_sentinel_lua)
    return _type(d) == "string" and #d > 0
end

local function _dump_reports_c(fn)
    if not _is_fn(_c_dump) or not _is_fn(fn) then return false end
    local d = _c_dump(fn)
    if d == nil then return true end
    if _type(d) == "string" and _match(d, "[Dd]ump") then return true end
    return false
end

local function _dump_cross_check(fn)
    if not _is_fn(_c_dump) then return true end
    if not _is_fn(fn) then return false end
    return _is_c_src(_c_info(fn, "s")) == _dump_reports_c(fn)
end

local function _no_hidden_upvalues(fn)
    if not _is_fn(fn) then return false end
    if not _isc(fn) then return true end
    if not _is_fn(_c_getup) then return true end
    return _c_getup(fn, 1) == nil
end

local function _info_line_ok(fn)
    if not _is_fn(fn) then return false end
    local line = _c_info(fn, "l")
    return line == nil or line == -1
end

local function _info_self_ok(fn)
    if not _is_fn(fn) then return false end
    return _c_info(fn, "f") == fn
end

local function _err_mentions(e, name)
    if _type(e) ~= "string" then return false end
    if not _match(e, "invalid argument") then return false end
    return _match(e, name) ~= nil
end

local function _error_fingerprint_ok()
    local e
    e = _c_ls(nil);       if not _err_mentions(e, "loadstring") then return false end
    e = _c_ls(true);      if not _err_mentions(e, "loadstring") then return false end
    e = _c_cc(nil);       if not _err_mentions(e, "coroutine")  then return false end
    e = _c_cr(nil);       if not _err_mentions(e, "coroutine")  then return false end
    e = _c_getup(nil, 1); if not _err_mentions(e, "getupvalue") then return false end
    if _is_fn(_c_setup) then
        e = _c_setup(nil, 1, nil)
        if not _err_mentions(e, "setupvalue") then return false end
    end
    e = _c_info(nil, "s"); if not _err_mentions(e, "getinfo") then return false end
    if _is_fn(_c_dump) then
        e = _c_dump(nil); if not _err_mentions(e, "dump") then return false end
    end
    return true
end

local function _err_shape_ok()
    local r1 = _c_ls({})
    if r1 ~= nil and (_type(r1) ~= "string" or #r1 < 3) then return false end
    if _type(_c_cc("not_a_function")) ~= "string" then return false end
    if _type(_c_cr("not_a_coroutine")) ~= "string" then return false end
    if _type(_c_getup("not_a_function", 1)) ~= "string" then return false end
    local marker = "ERRPROBE_a7f3c9b1"
    local e5 = _c_err(marker)
    if _type(e5) ~= "string" or not _match(e5, marker) then return false end
    return true
end

local function _anti_prehook_ok()
    if not _is_fn(_c_dump) then
        for _, fn in ipairs({_c_ls, _c_info, _c_getup, _pcall, _xpcall}) do
            if not _no_hidden_upvalues(fn) or not _info_line_ok(fn) or not _info_self_ok(fn) then return false end
        end
        return true
    end
    if not _dump_works() then return false end
    for _, fn in ipairs({_c_ls, _c_info, _c_getup, _c_dump, _c_cc, _c_cr, _c_err, _pcall, _xpcall}) do
        if fn and not _dump_reports_c(fn) then return false end
    end
    for _, fn in ipairs({_c_ls, _c_info, _c_getup, _c_dump, _pcall, _xpcall}) do
        if not _no_hidden_upvalues(fn) or not _info_line_ok(fn) or not _info_self_ok(fn) then return false end
    end
    return true
end

local function _upvalues_ok()
    if not _is_fn(_c_getup) then return false end
    local count, names, i = 0, {}, 1
    while true do
        if i > 64 then return false end
        local name, val = _c_getup(_probe_body, i)
        if name == nil then break end
        names[name] = val; count = count + 1; i = i + 1
    end
    if count < 4 then return false end
    if names["_c_ls"] ~= _c_ls or names["_c_info"] ~= _c_info then return false end
    if names["_c_wr"] ~= _c_wr or names["_probe_body"] ~= _probe_body then return false end
    return true
end

local _tick_prev = _tick()
local function _tick_ok()
    local t = _tick()
    if _type(t) ~= "number" or t < _tick_prev - 1 then return false end
    _tick_prev = t
    return true
end

_pcall(setmetatable, _canary_table, {

    end,

local _canary_ok
_canary_ok = function()
    if _honeypot_triggered then return false end
    if _canary_table._sentinel ~= "CANARY_9f3a2b1c" then return false end
    if _canary_func() ~= "CANARY_FUNC_8d4e2" then return false end
    if _is_fn(_c_getgc) then
        local ok, objs = _pcall(_c_getgc)
        if ok and _type(objs) == "table" then
            local found_t, found_f = false, false
            for _, o in pairs(objs) do
                if o == _canary_table then found_t = true end
                if o == _canary_func then found_f = true end
            end
            if not found_t or not found_f then return false end
        end
    end
    return true
end

-- ============================================================
-- C-STACK OVERFLOW PROBE (MULTI-CANAL CON XPCALL)
-- ============================================================
local _PROBE_DEPTH   = 198
local _MAX_DEPTH_MAX = 210

local function _detect_hook_via_cstack(fn)
    if not _is_fn(fn) then return false end
    local wrapped = fn
    for _ = 1, _PROBE_DEPTH do wrapped = _c_wr(wrapped) end
    
    -- Canal 1: pcall
    local ok_p, err_p = _pcall(wrapped)
    if not ok_p and _type(err_p) == "string" and _match(err_p, "C stack overflow") then return true end
    
    -- Canal 2: xpcall
    if _is_fn(_xpcall) then
        local ok_x, err_x = _xpcall(wrapped, function(e) return e end)
        if not ok_x and _type(err_x) == "string" and _match(err_x, "C stack overflow") then return true end
    end

    -- Canal 3: coroutine.resume (Bypass total de pcall/xpcall hooks)
    local co = _c_cc(wrapped)
    local ok_c, err_c = _c_cr(co)
    if not ok_c and _type(err_c) == "string" and _match(err_c, "C stack overflow") then return true end

    return false
end

local function _calibrate()
    local function ok(n) return _safe(function() _c_wr(_probe_body)(n) end) == true end
    local lo, hi, guard = 8, 16, 0
    while ok(hi) and hi < 240 do
        lo = hi; hi = hi * 2; guard = guard + 1
        if guard > 8 then break end
    end
    if not ok(hi) then
        while lo < hi do
            local mid = _floor((lo + hi + 1) / 2)
            if ok(mid) then lo = mid else hi = mid - 1 end
        end
        return lo
    end
    return hi
end

local _max_depth = _calibrate()

_timing_base = _calibrate_timing()
_gc_timing_base = _calibrate_gc_timing()
if _is_fn(_gcinfo) then _gc_start = _gcinfo() end

if _max_depth < _PROBE_DEPTH or _max_depth > _MAX_DEPTH_MAX then _b() return end

local _probe_depth = _PROBE_DEPTH
local _last_cal    = _tick()

local function _run_probe()
    return _safe(function() _c_wr(_probe_body)(_probe_depth) end) == true
end

local function _maybe_recalibrate()
    if _tick() - _last_cal < 60 then return end
    _last_cal = _tick()
    local new_max = _calibrate()
    if new_max < _PROBE_DEPTH or new_max > _MAX_DEPTH_MAX then _b() return end
end

local function _global_identity_ok()
    if _renv[_n_ls] ~= nil and _renv[_n_ls] ~= _c_ls then return false end
    if _renv[_n_dbg] ~= nil and _renv[_n_dbg] ~= _dbg then return false end
    if _renv[_n_co] ~= nil and _renv[_n_co] ~= _c_co then return false end
    if _renv[_n_pcall] ~= nil and _renv[_n_pcall] ~= _pcall then return false end
    if _renv[_n_xpcall] ~= nil and _renv[_n_xpcall] ~= _xpcall then return false end
    if _renv[_n_sp] ~= nil and _renv[_n_sp] ~= _snap_renv.sp then return false end
    return true
end

local function _singleton_ok()
    if not _is_fn(_c_getgc) then return true end
    local ok, objs = _pcall(_c_getgc)
    if not ok or _type(objs) ~= "table" then return true end
    local n_sv, n_rp = 0, 0
    for _, o in pairs(objs) do
        if o == _secure_validate then n_sv = n_sv + 1 end
        if o == _run_probe       then n_rp = n_rp + 1 end
    end
    return n_sv == 1 and n_rp == 1
end

local function _secure_validate()
    for _, k in ipairs({_n_pcall,_n_xpcall,_n_tostr,_n_type,_n_rawget,_n_isc,_n_tick}) do
        if _fenv[k] ~= _renv[k] then _b() return end
    end
    if _fenv[_n_fenv] ~= _renv[_n_fenv] or _fenv[_n_renv] ~= _renv[_n_renv] then _b() return end
    if _fenv[_n_setf] ~= _c_setf or _fenv[_n_sp] ~= _renv[_n_sp] then _b() return end
    if _fenv[_n_sp][_n_wait] ~= _c_wait then _b() return end
    if _fenv[_n_err] ~= _renv[_n_err] then _b() return end

    if _type(_fenv[_n_game]) ~= "userdata" or _fenv[_n_game] ~= _c_game then _b() return end
    if _type(_fenv[_n_work]) ~= "userdata" or _fenv[_n_work] ~= _c_work then _b() return end

    if not _global_identity_ok() then _b() return end

    if not _timing_ok()    then _b() return end
    if not _gc_ok()        then _b() return end
    if not _gc_timing_ok() then _b() return end
    if not _gc_hook_ok()   then _b() return end
    if not _canary_ok()    then _b() return end
    if not _env_leak_ok()  then _b() return end

    -- C-Stack Probes
    if _detect_hook_via_cstack(_c_ls)   then _b() return end
    if _detect_hook_via_cstack(_c_info) then _b() return end
    if _detect_hook_via_cstack(_c_wr)   then _b() return end
    if _detect_hook_via_cstack(_pcall)  then _b() return end
    if _detect_hook_via_cstack(_xpcall) then _b() return end
    if _is_fn(_c_dump) and _detect_hook_via_cstack(_c_dump) then _b() return end

    if not _ls_works()             then _b() return end
    if not _getup_works()          then _b() return end
    if not _err_shape_ok()         then _b() return end
    if not _error_fingerprint_ok() then _b() return end
    if not _closure_types_ok()     then _b() return end
    if not _upvalues_ok()          then _b() return end
    if not _anti_prehook_ok()      then _b() return end
    if not _proto_ok()             then _b() return end
    if not _safeenv_ok()           then _b() return end
    if not _stack_is_clean()       then _b() return end
    if not _singleton_ok()         then _b() return end

    if _c_meta then
        local _mt = _c_meta(_c_game)
        if _mt then
            local _nc = _rawget(_mt, _n_ncall)
            if _nc then
                if not _isc(_nc) or _c_info(_nc, "u") > 0 or not _no_hidden_upvalues(_nc) or not _info_line_ok(_nc) then _b() return end
            end
            local _idx = _rawget(_mt, _n_idx)
            if _idx and _type(_idx) == "function" then
                if not _isc(_idx) or _c_info(_idx, "u") > 0 or not _no_hidden_upvalues(_idx) or not _info_line_ok(_idx) then _b() return end
            end
        end
    end

    if not _tick_ok() then _b() return end
    return true
end

local function _wrap(orig)
    return function(...)
        if _state.breached then _c_err("sandbox_locked", 0) end
        if not _secure_validate() then _b() end
        return orig(...)
    end
end

_sandbox = {
    [_n_game]=_c_game, [_n_work]=_c_work, [_n_ls]=_wrap(_c_ls),
    [_n_sp]=_snap_renv.sp, [_n_tick]=_tick,
    [_n_pcall]=_pcall, [_n_xpcall]=_xpcall, [_n_tostr]=_tostr, [_n_type]=_type,
    [_n_str]=_renv[_n_str], [_n_math]=_renv[_n_math],
    table=_renv.table, coroutine=_renv[_n_co],
    print=_renv.print, warn=_renv.warn, error=_c_err,
    typeof=_renv.typeof,
}

_renv.setmetatable(_sandbox, { __metatable = false })

local _heartbeat_pool  = {}
local _heartbeat_count = 0

local function _rand_wait() return 0.12 + _rand() * 0.42 end

local function _spawn_heartbeat(id)
    _c_sp(function()
        while not _state.breached do
            if not _secure_validate() then _b() return end
            if _state.breached then return end
            if not _run_probe() then _b() return end
            _maybe_recalibrate()
            if _state.breached then return end
            _heartbeat_pool[id] = _tick()
            _c_wait(_rand_wait())
        end
    end)
end

for i = 1, 3 do
    _spawn_heartbeat(i)
    _heartbeat_count = _heartbeat_count + 1
end

_c_sp(function()
    local stalls = 0
    while not _state.breached do
        _c_wait(0.5 + _rand() * 0.25)
        if _state.breached then return end
        local now = _tick()
        local active = 0
        for _, last_time in pairs(_heartbeat_pool) do
            if now - last_time < 2.0 then active = active + 1 end
        end
        if active < _heartbeat_count then
            stalls = stalls + 1
            if stalls >= 2 then _b() return end
        else
            stalls = 0
        end
        if not _secure_validate() then return end
    end
end)
